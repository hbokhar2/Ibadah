set(TARGET_NAME Systems)

file(GLOB INPUT_MANAGER_SOURCES CONFIGURE_DEPENDS
	"${CMAKE_CURRENT_SOURCE_DIR}/InputManager/*.cpp"
)

file(GLOB CONFIG_MANAGER_SOURCES CONFIGURE_DEPENDS
	"${CMAKE_CURRENT_SOURCE_DIR}/ConfigManager/*.cpp"
)

file(GLOB ESCAPE_MENU_SOURCES CONFIGURE_DEPENDS
	"${CMAKE_CURRENT_SOURCE_DIR}/EscapeMenu/*.cpp"
)

file(GLOB SYSTEM_SOURCES CONFIGURE_DEPENDS
	"${CMAKE_CURRENT_SOURCE_DIR}/*.cpp"
)

add_library(${TARGET_NAME} STATIC
	${INPUT_MANAGER_SOURCES}
	${CONFIG_MANAGER_SOURCES}
	${ESCAPE_MENU_SOURCES}
	${SYSTEM_SOURCES}
)

set_target_properties(${TARGET_NAME} PROPERTIES POSITION_INDEPENDENT_CODE ON)

target_include_directories(${TARGET_NAME} PUBLIC
	${CMAKE_CURRENT_SOURCE_DIR}
	${CMAKE_CURRENT_SOURCE_DIR}/InputManager
	${CMAKE_CURRENT_SOURCE_DIR}/EscapeMenu
)

target_link_libraries(${TARGET_NAME} PUBLIC
	godot-cpp
)
